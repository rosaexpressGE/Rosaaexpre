      });

      // Ordenar por fecha de creación (más recientes primero)
      list.sort((a, b) => {
        const timeA = new Date(a.fechaCreacion || a.createdAt || a.fecha || 0).getTime();
        const timeB = new Date(b.fechaCreacion || b.createdAt || b.fecha || 0).getTime();
        return timeB - timeA;
      });

      window.allSolicitudes = list;
      window.solicitudes = list.filter(s => !s.eliminado);

    function copiarTextoAlPortapapeles(texto) {
      if (navigator.clipboard && window.isSecureContext) {
        navigator.clipboard.writeText(texto);
      } else {
        const textarea = document.createElement('textarea');
        textarea.value = texto;
        textarea.style.position = 'fixed';
        textarea.style.left = '-9999px';
        document.body.appendChild(textarea);
        textarea.focus();
        textarea.select();
        try {
          document.execCommand('copy');
        } catch (err) {
          console.error('Fallback copy error:', err);
        }
        document.body.removeChild(textarea);
      }
    }
